input scanMode = {default BullishSetup, BullishWatch, BearishSetup, BearishWatch, NeutralCompression};
input length = 20;
input fast_ema_length = 8;
input slow_ema_length = 21;
input bb_mult = 2.0;
input kc_mult = 1.5;
input withinBars = 1;
input useMacroFilter = yes;
input macro_sma_fast = 50;
input macro_sma_slow = 200;
input bullish_near_min_pct = -1.0;
input bullish_near_max_pct = 2.0;
input bearish_near_min_pct = -2.0;
input bearish_near_max_pct = 1.0;

def avg = SimpleMovingAvg(close, length);
def sDev = StDev(close, length);
def upperBB = avg + (bb_mult * sDev);
def lowerBB = avg - (bb_mult * sDev);
def trueRange = TrueRange(high, close, low);
def avg_true_range = SimpleMovingAvg(trueRange, length);
def upperKC = avg + (kc_mult * avg_true_range);
def lowerKC = avg - (kc_mult * avg_true_range);
def compressionActive = lowerBB > lowerKC and upperBB < upperKC;
def fastEma = ExpAverage(close, fast_ema_length);
def slowEma = ExpAverage(close, slow_ema_length);
def sma50 = SimpleMovingAvg(close, macro_sma_fast);
def sma200 = SimpleMovingAvg(close, macro_sma_slow);
def shortTrendUp = fastEma > slowEma;
def shortTrendDown = fastEma < slowEma;
def macroTrendUp = if useMacroFilter then close > sma50 and sma50 > sma200 else yes;
def macroTrendDown = if useMacroFilter then close < sma50 and sma50 < sma200 else yes;
def trendUp = shortTrendUp and macroTrendUp;
def trendDown = shortTrendDown and macroTrendDown;
def percentFromSlowEma = ((close - slowEma) / slowEma) * 100;
def nearBullishSlowEma = percentFromSlowEma >= bullish_near_min_pct and percentFromSlowEma <= bullish_near_max_pct;
def nearBearishSlowEma = percentFromSlowEma >= bearish_near_min_pct and percentFromSlowEma <= bearish_near_max_pct;
def bullishSetup = compressionActive and trendUp and nearBullishSlowEma;
def bullishWatch = compressionActive and trendUp and !nearBullishSlowEma;
def bearishSetup = compressionActive and trendDown and nearBearishSlowEma;
def bearishWatch = compressionActive and trendDown and !nearBearishSlowEma;
def neutralCompression = compressionActive and !trendUp and !trendDown;
def bullishSetupRecent = bullishSetup within withinBars bars;
def bullishWatchRecent = bullishWatch within withinBars bars;
def bearishSetupRecent = bearishSetup within withinBars bars;
def bearishWatchRecent = bearishWatch within withinBars bars;
def neutralCompressionRecent = neutralCompression within withinBars bars;

plot scanSignal =
    if scanMode == scanMode.BullishSetup then bullishSetupRecent
    else if scanMode == scanMode.BullishWatch then bullishWatchRecent
    else if scanMode == scanMode.BearishSetup then bearishSetupRecent
    else if scanMode == scanMode.BearishWatch then bearishWatchRecent
    else neutralCompressionRecent;
